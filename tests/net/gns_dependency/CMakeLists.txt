cmake_minimum_required(VERSION 3.25)
project(EawrGnsDependencySmoke LANGUAGES CXX)
include(CTest)
set(EAWR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
include("${EAWR_ROOT}/cmake/EawrReserveGuard.cmake")
include("${EAWR_ROOT}/cmake/EawrGns.cmake")
option(EAWR_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
include("${EAWR_ROOT}/cmake/EawrWarnings.cmake")
add_executable(gns_dependency_smoke smoke.cpp)
if(EAWR_ENABLE_GNS)
    if(WIN32)
        target_sources(gns_dependency_smoke PRIVATE "${EAWR_ROOT}/src/net/platform_socket_win.cpp")
        target_link_libraries(gns_dependency_smoke PRIVATE ws2_32)
    else()
        target_sources(gns_dependency_smoke PRIVATE "${EAWR_ROOT}/src/net/platform_socket_posix.cpp")
    endif()
endif()
target_compile_features(gns_dependency_smoke PRIVATE cxx_std_20)
target_link_libraries(gns_dependency_smoke PRIVATE eawr::gns_dependency)
eawr_target_project_warnings(gns_dependency_smoke)
add_test(NAME net_gns_dependency_smoke COMMAND gns_dependency_smoke)
set_tests_properties(net_gns_dependency_smoke PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
