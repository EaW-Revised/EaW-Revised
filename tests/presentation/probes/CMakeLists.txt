# One binding/core build serves every standalone GDExtension. Discover their
# declarations so a new probe project automatically joins the offload gate.
cmake_minimum_required(VERSION 3.25)
project(EawrProbes LANGUAGES CXX)
get_filename_component(EAWR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${EAWR_ROOT}/cmake/EawrProbeSources.cmake")
file(GLOB_RECURSE probe_projects CONFIGURE_DEPENDS "${EAWR_ROOT}/tests/presentation/*/CMakeLists.txt")
set(probe_targets)
foreach(probe_project IN LISTS probe_projects)
    file(READ "${probe_project}" contents)
    string(REGEX MATCHALL "add_library\\([A-Za-z0-9_]+_probe SHARED" declarations "${contents}")
    if(declarations)
        get_filename_component(directory "${probe_project}" DIRECTORY)
        file(RELATIVE_PATH relative "${EAWR_ROOT}/tests/presentation" "${directory}")
        add_subdirectory("${directory}" "${relative}")
        foreach(declaration IN LISTS declarations)
            string(REGEX REPLACE "add_library\\(([A-Za-z0-9_]+) SHARED" "\\1" target "${declaration}")
            list(APPEND probe_targets "${target}")
        endforeach()
    endif()
endforeach()
if(NOT probe_targets)
    message(FATAL_ERROR "No standalone probe libraries discovered")
endif()
add_custom_target(eawr_probes ALL DEPENDS ${probe_targets})
