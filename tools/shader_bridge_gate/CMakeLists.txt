cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(EawrShaderBridgeGate LANGUAGES CXX)
    option(EAWR_ENABLE_SHADER_BRIDGE_GATE "Build the external-DLL shader bridge experiment" OFF)
endif()
if(NOT EAWR_ENABLE_SHADER_BRIDGE_GATE)
    return()
endif()
if(NOT WIN32)
    message(FATAL_ERROR "The shader bridge gate exercises the Windows dynamic DLL boundary")
endif()
set(EAWR_VKD3D_INCLUDE_DIR "" CACHE PATH "External unmodified vkd3d public headers")
if(NOT EXISTS "${EAWR_VKD3D_INCLUDE_DIR}/vkd3d_shader.h")
    message(FATAL_ERROR "Set EAWR_VKD3D_INCLUDE_DIR to the external release's include directory")
endif()
add_executable(eawr_shader_bridge_gate main.cpp)
target_compile_features(eawr_shader_bridge_gate PRIVATE cxx_std_20)
target_include_directories(eawr_shader_bridge_gate SYSTEM PRIVATE "${EAWR_VKD3D_INCLUDE_DIR}")
if(MSVC)
    target_compile_options(eawr_shader_bridge_gate PRIVATE /W4 /WX /permissive- /EHsc)
else()
    target_compile_options(eawr_shader_bridge_gate PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
