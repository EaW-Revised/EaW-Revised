cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(EawrRelayServer LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    option(EAWR_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/EawrWarnings.cmake")
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/EawrReserveGuard.cmake")
    set(EAWR_ENABLE_RELAY ON CACHE BOOL "" FORCE)
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/EawrRelay.cmake")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../src/net/relay_sources.cmake")
add_library(eawr_relay_service STATIC ${EAWR_RELAY_SERVICE_SOURCES})
target_link_libraries(eawr_relay_service PUBLIC eawr::relay)
eawr_target_project_warnings(eawr_relay_service)
add_executable(eawr_relay_server main.cpp)
target_include_directories(eawr_relay_server PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src/net")
target_link_libraries(eawr_relay_server PRIVATE eawr_relay_service)
eawr_target_project_warnings(eawr_relay_server)
install(TARGETS eawr_relay_server RUNTIME DESTINATION bin)
